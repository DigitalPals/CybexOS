cmake_minimum_required(VERSION 3.16)
project(CybexOSNotificationImages VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Gui Qml Quick)
include(CTest)
set(NOTIFICATION_IMAGES_QML_DIR "lib64/qt6/qml/CybexOS/Images" CACHE STRING "Installed QML module directory")
add_library(notification-image-fingerprint STATIC imagefingerprints.cpp imagefingerprints.h)
set_target_properties(notification-image-fingerprint PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(notification-image-fingerprint PUBLIC Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick)
add_library(cybexosnotificationimagesplugin MODULE plugin.cpp)
target_link_libraries(cybexosnotificationimagesplugin PRIVATE notification-image-fingerprint)
set_target_properties(cybexosnotificationimagesplugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/qml/CybexOS/Images")
configure_file(qmldir qml/CybexOS/Images/qmldir COPYONLY)
configure_file(plugins.qmltypes qml/CybexOS/Images/plugins.qmltypes COPYONLY)
install(TARGETS cybexosnotificationimagesplugin LIBRARY DESTINATION "${NOTIFICATION_IMAGES_QML_DIR}")
install(FILES qmldir plugins.qmltypes DESTINATION "${NOTIFICATION_IMAGES_QML_DIR}")
install(FILES LICENSE DESTINATION share/licenses/cybexos-notification-images)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(notification-image-fingerprint-test tests.cpp)
    target_link_libraries(notification-image-fingerprint-test PRIVATE notification-image-fingerprint Qt6::Test)
    target_compile_definitions(notification-image-fingerprint-test PRIVATE TEST_QML_PATH="${CMAKE_BINARY_DIR}/qml")
    add_dependencies(notification-image-fingerprint-test cybexosnotificationimagesplugin)
    add_test(NAME notification-image-fingerprint COMMAND notification-image-fingerprint-test)
    set_tests_properties(notification-image-fingerprint PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
